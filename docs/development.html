開発者向けドキュメント

Syncridorは、Pythonとpygameを使用したモジュラー設計のボードゲーム実装です。このドキュメントでは、コード構造、API、テスト、拡張方法について詳しく説明します。

🏗️ アーキテクチャ概要

システム構成

Syncridor システム構成
├── ゲームロジック層 (SyncridorGame)
│   ├── 盤面状態管理
│   ├── ルール検証エンジン
│   ├── AI戦略実装
│   └── パスファインディング

  • 関心の分離: ゲームロジックとUIを完全分離

Wall クラス

壁オブジェクトのデータクラス

class Wall:
    def __init__(self, row: int, col: int, is_horizontal: bool, player: Player = None):
        """
        壁を初期化
        
        Args:
            row (int): 行位置 (0-7)
            col (int): 列位置 (0-7)  
            is_horizontal (bool): True=水平, False=垂直
            player (Player): 配置したプレイヤー
        """
        self.row = row
        self.col = col
        self.is_horizontal = is_horizontal
        self.player = player

    def __eq__(self, other):
        """壁の等価性を判定（位置と方向が同じ）"""
        return (self.row == other.row and 
                self.col == other.col and 
                self.is_horizontal == other.is_horizontal)

イベント処理

def handle_events(self):
    """pygame イベントの処理"""
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            return False
            
        elif event.type == pygame.MOUSEBUTTONDOWN:
            if event.button == 1:  # 左クリック = 移動
                self.handle_player_move(event.pos)
            elif event.button == 3:  # 右クリック = 壁配置
                self.handle_wall_placement(event.pos)
                
        elif event.type == pygame.KEYDOWN:
            if event.key == pygame.K_SPACE:  # 壁の向き切り替え
                self.horizontal_wall = not self.horizontal_wall
            elif event.key == pygame.K_r:  # リスタート
                self.game.reset_game()
            elif event.key == pygame.K_m:  # メインメニュー
                self.show_menu = True

class TestPlayerMovement(unittest.TestCase):
    def setUp(self):
        self.game = SyncridorGame(GameMode.HUMAN_VS_HUMAN)
    
    def test_basic_movement(self):
        """基本的な移動テスト"""
        initial_pos = self.game.player1_pos
        new_pos = (initial_pos[0] - 1, initial_pos[1])
        
        result = self.game.move_player(new_pos)
        self.assertTrue(result)
        self.assertEqual(self.game.player1_pos, new_pos)
        self.assertEqual(self.game.current_player, Player.PLAYER2)

2. 戦略メソッドの実装

def ai_make_move_minimax(self):
    """ミニマックス法による戦略"""
    def minimax(position, depth, maximizing_player):
        if depth == 0 or self.game_over:
            return self.evaluate_position(position)
        
        if maximizing_player:
            max_eval = float('-inf')
            for move in self.get_possible_moves():
                eval_score = minimax(move, depth-1, False)
                max_eval = max(max_eval, eval_score)
            return max_eval
        else:
            min_eval = float('inf')
            for move in self.get_opponent_moves():
                eval_score = minimax(move, depth-1, True)
                min_eval = min(min_eval, eval_score)
            return min_eval
    
    best_move = None
    best_score = float('-inf')
    
    for move in self.get_possible_moves():
        score = minimax(move, depth=3, maximizing_player=False)
        if score > best_score:
            best_score = score
            best_move = move
    
    return self.execute_move(best_move)

3. 戦略の登録

def ai_make_move(self):
    """現在のプレイヤーのAI戦略に応じて行動を決定"""
    strategy = (self.player1_strategy if self.current_player == Player.PLAYER1 
                else self.player2_strategy)
    
    if strategy == AIStrategy.AGGRESSIVE:
        return self.ai_make_move_aggressive()
    elif strategy == AIStrategy.DEFENSIVE:
        return self.ai_make_move_defensive()
    elif strategy == AIStrategy.BALANCED:
        return self.ai_make_move_balanced()
    elif strategy == AIStrategy.MINIMAX:
        return self.ai_make_move_minimax()      # 新しい戦略
    elif strategy == AIStrategy.LEARNED:
        return self.ai_make_move_learned()      # 機械学習戦略

def handle_tournament_mode(self):
    """トーナメントモードの処理"""
    if not hasattr(self, 'tournament_matches'):
        self.tournament_matches = self.setup_tournament()
    
    current_match = self.tournament_matches[self.current_match_index]
    
    if self.game_over:
        self.record_tournament_result(current_match, self.winner)
        self.advance_tournament()
    
    return self.continue_tournament()

パフォーマンス測定

import cProfile
import pstats

def profile_ai_performance():
    """AI戦略のパフォーマンスを測定"""
    profiler = cProfile.Profile()
    
    # AIの実行をプロファイル
    profiler.enable()
    game = SyncridorGame(GameMode.AI_VS_AI)
    
    for _ in range(100):  # 100回の行動
        game.ai_make_move()
        
    profiler.disable()
    
    # 結果の分析
    stats = pstats.Stats(profiler)
    stats.sort_stats('cumulative')
    stats.print_stats(10)  # 上位10関数

📚 コーディング規約

命名規則

  • クラス名: PascalCase (例: SyncridorGame)
  • 関数・変数名: snake_case (例: get_shortest_path)
  • 定数: UPPER_SNAKE_CASE (例: BOARD_SIZE)
  • プライベートメソッド: 先頭にアンダースコア (例: _calculate_score)

docstring 規則

def get_shortest_path(self, start_pos, goal_row, walls=None):
    """BFS（幅優先探索）でゴールまでの最短経路を計算
    
    Args:
        start_pos (tuple[int, int]): 開始位置 (row, col)
        goal_row (int): ゴール行（プレイヤー1: 0, プレイヤー2: 8）
        walls (list[Wall], optional): 考慮する壁のリスト. Defaults to None.
    
    Returns:
        list[tuple[int, int]]: 最短経路のリスト [(row, col), ...]
        空リストが返される場合は経路なし
        
    Raises:
        ValueError: start_pos が無効な位置の場合
        
    Example:
        >>> game = SyncridorGame(GameMode.HUMAN_VS_HUMAN)
        >>> path = game.get_shortest_path((8, 4), 0)
        >>> print(f"Path length: {len(path)}")
        Path length: 9
    """
